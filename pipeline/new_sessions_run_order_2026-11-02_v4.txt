import pandas as pd
from pandas.api.types import is_numeric_dtype as isnum
ign={('sub-008',2),('sub-010',3),('sub-092',2),('sub-108',2)}
for f in ['univariate_v1_harmonized_sqrt.csv','rsa_v1_harmonized.csv']:
    o=pd.read_csv('outputs/D_liu.bak_YYYYMMDD/'+f); n=pd.read_csv('outputs/D_liu/'+f)
    for d in (o,n): d['ses_n']=d.session.astype(str).str.replace('ses-','').astype(float).astype(int)
    keys=[c for c in o.columns if not isnum(o[c])]+['ses_n']
    num=[c for c in o.columns if isnum(o[c]) and c not in ('session','ses_n') and c in n.columns]
    m=o.merge(n,on=keys,suffixes=('_o','_n')); m=m[[(s,int(x)) not in ign for s,x in zip(m.subject_id,m.ses_n)]]
    print('==',f,len(m),'rows compared')
    for c in num:
        x=(m[c+'_n']-m[c+'_o']).abs()
        if x.max()>0: print(f'  {c:22s} max {x.max():.3f}  median {x.median():.4f}  changed>0.1: {(x>0.1).sum()}/{x.notna().sum()}')
EOF
